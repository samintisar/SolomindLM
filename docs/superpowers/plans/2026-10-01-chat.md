import { useCallback, useEffect, useMemo, useRef, useState } from "react";
import { Popover, PopoverAnchor, PopoverContent } from "@/shared/components/ui/popover";
import type { ReferenceChunk } from "@/shared/types/index";
import type { RefHandlers } from "../utils/messageRendering.utils";
import { CitationCard } from "./CitationCard";

const OPEN_DELAY_MS = 80;
const CLOSE_DELAY_MS = 150;

interface Target {
  refId: number;
  messageId: string;
  el: HTMLElement;
}

interface Options {
  resolveReference: (messageId: string, refId: number) => ReferenceChunk | null;
  onOpenInSources?: (ref: ReferenceChunk) => (() => void) | undefined;
  onAddToNotebook?: (ref: ReferenceChunk) => (() => void) | undefined;
}

/**
 * One popover per chat panel, anchored to whichever citation chip is active. A per-chip trigger
 * would break under Virtuoso, which unmounts off-screen rows.
 */
export function useCitationPopover({ resolveReference, onOpenInSources, onAddToNotebook }: Options) {
  const [target, setTarget] = useState<Target | null>(null);
  const openTimer = useRef<ReturnType<typeof setTimeout>>(undefined);
  const closeTimer = useRef<ReturnType<typeof setTimeout>>(undefined);

  const clearTimers = useCallback(() => {
    clearTimeout(openTimer.current);
    clearTimeout(closeTimer.current);
  }, []);
  const scheduleClose = useCallback(() => {
    clearTimeout(openTimer.current);
    clearTimeout(closeTimer.current);
    closeTimer.current = setTimeout(() => setTarget(null), CLOSE_DELAY_MS);
  }, []);

  useEffect(() => clearTimers, [clearTimers]);

  const handlers = useMemo<RefHandlers>(
    () => ({
      onRefEnter: (refId, messageId, el) => {
        clearTimers();
        openTimer.current = setTimeout(() => setTarget({ refId, messageId, el }), OPEN_DELAY_MS);
      },
      onRefLeave: scheduleClose,
      onRefToggle: (refId, messageId, el) => {
        clearTimers();
        setTarget((t) =>
          t && t.refId === refId && t.messageId === messageId ? null : { refId, messageId, el }
        );
      },
    }),
    [clearTimers, scheduleClose]
  );

  const reference = target ? resolveReference(target.messageId, target.refId) : null;
  const anchorRef = useMemo(() => ({ current: target?.el ?? null }), [target]);

  const popover = (
    <Popover open={!!(target && reference)} onOpenChange={(open) => !open && setTarget(null)}>
      <PopoverAnchor virtualRef={anchorRef as React.RefObject<HTMLElement>} />
      {reference && (
        <PopoverContent
          side="top"
          align="center"
          sideOffset={6}
          collisionPadding={16}
          className="w-96 p-0"
          onOpenAutoFocus={(e) => e.preventDefault()}
          onPointerEnter={clearTimers}
          onPointerLeave={(e) => e.pointerType === "mouse" && scheduleClose()}
          onInteractOutside={(e) => {
            // Clicking another chip retargets instead of closing.
            if ((e.target as Element).closest('[title^="Reference "]')) e.preventDefault();
          }}
        >
          <CitationCard
            reference={reference}
            onOpenInSources={onOpenInSources?.(reference)}
            onAddToNotebook={onAddToNotebook?.(reference)}
          />
        </PopoverContent>
      )}
    </Popover>
  );

  return { handlers, popover, close: () => setTarget(null) };
}
